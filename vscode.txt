git:
    git config pull.rebase false            # merge vai rebase default operation, vscode ei syncaa muuttaminen
    git checkout -b feature/blaa            # create a new feature specific branch (-b creates a branch)
    git checkout -b master
    git branch                              # list local branches
    git branch -a                           # list global branches
    git branch -d name                      # removes a local branch (blocks if unmerged)
    git branch -D name                      # removes a local branch (forced, data loss)
    git push origin --delete name           # removes a branch on remote
    git fetch --prune                       # Prune stale remote branches
